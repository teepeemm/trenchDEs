\section{Eigenvalue Problems for \texorpdfstring{$y''+\lambda y=0$}{y''+𝜆y=0}}
\label{sec:eig_probs}

In \zcref{ch:fourier_pde} we'll study partial differential equations
that arise in problems of heat conduction, wave propagation, and
potential theory. The purpose of this chapter is to develop tools
required to solve these equations. In this section we consider the
following problems, where $\lambda$ is a real number and $L>0$:

\begin{enumext}[label=\arabic*,wrap-label={Problem #1:},labelwidth=6em]
\item $y''+\lambda y=0,\quad y(0)=0,\quad y(L)=0$
\item $y''+\lambda y=0,\quad y'(0)=0,\quad y'(L)=0$
\item $y''+\lambda y=0,\quad y(0)=0,\quad y'(L)=0$
\item $y''+\lambda y=0,\quad y'(0)=0,\quad y(L)=0$
\item $y''+\lambda y=0,\quad y(-L)=y(L),\quad y'(-L)=y'(L)$
\end{enumext}

In each problem the conditions following the differential equation
are called \emph{\color{blue}boundary conditions}.
Note that the boundary conditions in Problem~5, unlike those in
Problems~1--4, don't require that $y$ or $y'$ be zero at the boundary
points, but only that $y$ have the same value at $x=\pm L$ , and that
$y'$ have the same value at $x=\pm L$. We say that the boundary
conditions in Problem~5 are \emph{\color{blue}periodic}.

Obviously, $y\equiv0$ (the trivial solution) is a solution of
Problems~1--5 for any value of $\lambda$. For most values of $\lambda$,
there are no other solutions. The interesting question is this:

\emph{\color{blue}For what values of $\lambda$ does the problem have
nontrivial solutions, and what are they?}

A value of $\lambda$ for which the problem has
a nontrivial solution is an \emph{\color{blue}eigenvalue} of the problem, and the
nontrivial solutions are $\lambda$-\emph{\color{blue}eigenfunctions}, or \emph{\color{blue}eigenfunctions associated with $\lambda$}. Note that a nonzero
constant multiple of a $\lambda$-eigenfunction is again a
$\lambda$-eigenfunction.

Problems~1--5 are called \emph{\color{blue}eigenvalue problems}. \emph{\color{blue}Solving}
an eigenvalue problem means finding all its eigenvalues and associated
eigenfunctions. We'll take it as given here that all the eigenvalues
of Problems~1--5 are real numbers. This is proved in a more general
setting in \zcref{sec:sturm_liouville}.


\begin{theorem}\label{thmtype:11.1.1}
Problems 1--5 have no negative eigenvalues. Moreover, $\lambda=0$
is an eigenvalue of Problems 2 and 5, with associated
eigenfunction $y_0=1$, but $\lambda=0$ isn't  an eigenvalue of
Problems~1, 3, or 4.
\end{theorem}

\begin{proof}
We  consider Problems 1--4, and leave Problem 5 to you
(\zcref{exer:11.1.1}). If
$y''+\lambda y=0$, then
$y(y''+\lambda y)=0$, so
\[
\int_0^L y(x)(y''(x)+\lambda y(x))\,dx=0;
\]
   therefore,
\begin{equation} \label{eq:11.1.1}
\lambda\int_0^L y^2(x)\,dx=-\int_0^L y(x)y''(x)\, dx.
\end{equation}
Integration by parts yields
\begin{equation} \label{eq:11.1.2}
\begin{aligned}
\int_0^L y(x)y''(x)\, dx
&=y(x)y'(x)\lims 0L-\int_0^L(y'(x))^2\,dx\\
&=y(L)y'(L)-y(0)y'(0)-\int_0^L (y'(x))^2\,dx.
\end{aligned}
\end{equation}
However, if $y$ satisfies any of the boundary conditions of Problems
1--4, then
\[
y(L)y'(L)-y(0)y'(0)=0;
\]
hence, \eqref{eq:11.1.1} and \eqref{eq:11.1.2} imply that
\[
\lambda\int_0^L y^2(x)\,dx=\int_0^L (y'(x))^2\, dx.
\]
If $y\not\equiv0$, then
$\int_0^L y^2(x)\,dx>0$. Therefore
 $\lambda\ge0$ and,  if  $\lambda=0$, then $y'(x)=0$ for all $x$
in
$(0,L)$ (why?), and  $y$ is constant on $(0,L)$.
Any constant function satisfies the boundary conditions of Problem~2,
so $\lambda=0$ is an eigenvalue of Problem~2 and any nonzero constant
function is an associated eigenfunction. However, the only constant
function that satisfies the boundary conditions of Problems $1$, $3$, or
$4$ is $y\equiv0$. Therefore $\lambda=0$ isn't  an eigenvalue of any of
these problems.
\end{proof}

\begin{example}[Problem 1]\label{example:11.1.1}
Solve the  eigenvalue problem
\begin{equation} \label{eq:11.1.3}
y''+\lambda y=0,\quad y(0)=0,\quad y(L)=0.
\end{equation}
\end{example}

\begin{solution}
From \zcref{thmtype:11.1.1}, any eigenvalues of \eqref{eq:11.1.3} must be
positive. If $y$ satisfies \eqref{eq:11.1.3} with $\lambda>0$, then
\[
y=c_1\cos\sqrt\lambda\, x+c_2\sin\sqrt\lambda\, x,
\]
where $c_1$ and $c_2$ are constants. The boundary condition $y(0)=0$
implies that $c_1=0$. Therefore $y=c_2\sin\sqrt\lambda\, x$. Now the
boundary condition $y(L)=0$ implies that $c_2\sin\sqrt\lambda\, L=0$.
To make $c_2\sin\sqrt\lambda\, L=0$ with $c_2\ne0$, we must choose
$\sqrt\lambda=n\pi/L$, where $n$ is a positive integer. Therefore
$\lambda_n=n^2\pi^2/L^2$ is an eigenvalue and
\[
y_n=\sin\frac{n\pi x}{L}
\]
is an associated eigenfunction.
\end{solution}

For future reference, we state the result of \zcref{example:11.1.1}
as a theorem.

\begin{theorem}\color{blue}\label{thmtype:11.1.2}
The eigenvalue problem
\[
y''+\lambda y=0,\quad y(0)=0,\quad y(L)=0
\]
has infinitely many positive eigenvalues
$\lambda_n=n^2\pi^2/L^2$,
with associated eigenfunctions
\[
y_n=\sin \frac{n\pi x}{L},\quad
n=1,2,3,\dots.
\]
There are no other eigenvalues.
\end{theorem}

\]
has the eigenvalue $\lambda_0=0$, with associated eigenfunction

\begin{example}[Problem 3] \label{example:11.1.2}
Solve the
eigenvalue problem
\begin{equation} \label{eq:11.1.4}
y''+\lambda y=0,\quad y(0)=0,\quad y'(L)=0.
\end{equation}
\end{example}

\begin{solution}
From \zcref{thmtype:11.1.1},
 any eigenvalues of \eqref{eq:11.1.4} must be  positive.
If $y$ satisfies \eqref{eq:11.1.4} with  $\lambda>0$, then
\[
y=c_1\cos\sqrt\lambda\, x+c_2\sin\sqrt\lambda\, x,
\]
where $c_1$ and $c_2$ are constants. The boundary condition $y(0)=0$
implies that $c_1=0$. Therefore $y=c_2\sin\sqrt\lambda\, x$. Hence,
$y'=c_2\sqrt\lambda\cos\sqrt\lambda\,x$ and the boundary condition
$y'(L)=0$ implies that $c_2\cos\sqrt\lambda\,L=0$. To make
$c_2\cos\sqrt\lambda\,L=0$ with $c_2\ne0$ we must choose
\[
\sqrt\lambda=\frac{2n-1}{2L},
\]
where $n$ is a positive integer. Then $\lambda_n=(2n-1)^2\pi^2/4L^2$
is an eigenvalue and
\[
y_n=\sin\frac{(2n-1)\pi x}{2L}
\]
is an associated eigenfunction.
\end{solution}

For future reference, we state the result of \zcref{example:11.1.2}
as a theorem.

y''+\lambda y=0,\quad y(0)=0,\quad y'(L)=0

has infinitely many positive eigenvalues
$\lambda_n=(2n-1)^2\pi^2/4L^2$, with associated eigenfunctions

The eigenvalue problem

\]
has infinitely many positive eigenvalues

\begin{example}[Problem 5]\label{example:11.1.3}
Solve the eigenvalue  problem
\begin{equation} \label{eq:11.1.5}
y''+\lambda y=0,\quad y(-L)=y(L),\quad y'(-L)=y'(L).
\end{equation}
\end{example}

\begin{solution}
From \zcref{thmtype:11.1.1}, $\lambda=0$ is an eigenvalue of
\eqref{eq:11.1.5} with associated eigenfunction $y_0=1$, and any other
eigenvalues must be positive. If $y$ satisfies \eqref{eq:11.1.5} with
$\lambda>0$, then
\begin{equation} \label{eq:11.1.6}
y=c_1\cos\sqrt\lambda\, x+c_2\sin\sqrt\lambda\, x,
\end{equation}
where $c_1$ and $c_2$ are constants.
The boundary condition $y(-L)=y(L)$ implies that
\begin{equation} \label{eq:11.1.7}
c_1\cos(-\sqrt\lambda\,L)+c_2\sin(-\sqrt\lambda\,L)=c_1\cos \sqrt\lambda\,L+c_2\sin \sqrt\lambda\,L.
\end{equation}
Since
\begin{equation} \label{eq:11.1.8}
\cos(-\sqrt\lambda\,L)=\cos \sqrt\lambda\,L\quad\text{and}\quad\sin(-\sqrt\lambda\,L)=-\sin \sqrt\lambda\,L,
\end{equation}
\eqref{eq:11.1.7} implies that
\begin{equation} \label{eq:11.1.9}
c_2\sin \sqrt\lambda\,L=0.
\end{equation}
Differentiating \eqref{eq:11.1.6} yields
\[
y'=\sqrt\lambda\left(-c_1\sin\sqrt\lambda x+c_2\cos\sqrt\lambda
x\right).
\]
The boundary condition $y'(-L)=y'(L)$  implies that
\[
-c_1\sin(-\sqrt\lambda\,L)+c_2\cos(-\sqrt\lambda\,L)=-c_1\sin \sqrt\lambda\,L+c_2\cos \sqrt\lambda\,L,
\]
and  \eqref{eq:11.1.8} implies that
\begin{equation} \label{eq:11.1.10}
c_1\sin \sqrt\lambda\,L=0.
\end{equation}

\zcref{eq:11.1.9,eq:11.1.10} imply that $c_1=c_2=0$ unless
$\sqrt\lambda =n\pi /L$, where $n$ is a positive integer. In this  case
\eqref{eq:11.1.9} and \eqref{eq:11.1.10} both hold for arbitrary $c_1$ and
$c_2$. The eigenvalue determined in this way is
$\lambda_n=n^2\pi^2/L^2$, and each such eigenvalue has the linearly
independent associated eigenfunctions
\[
\cos \frac{n\pi x}{L}
\quad\text{and}\quad
  \sin\frac{ n\pi x}{L}.\qedhere
\]
\end{solution}

For future reference we state the result of \zcref{example:11.1.3} as
a theorem.


\begin{theorem}\label{thmtype:11.1.6}
The eigenvalue problem
\[
y''+\lambda y=0,\quad y(-L)=y(L),\quad y'(-L)=y'(L),
\]
has the eigenvalue $\lambda_0=0$, with associated eigenfunction $y_0=1$
and  infinitely many positive eigenvalues
$\lambda_n=n^2\pi^2/L^2$,
with associated eigenfunctions
\[
y_{1n}=\cos \frac{n\pi x}{L}
\quad\text{and}\quad
y_{2n}=\sin \frac{n\pi x}{L},\quad n=1,2,3,\dots.
\]
There are no other eigenvalues.
\end{theorem}


\boxit{Orthogonality}

We say that two integrable functions $f$ and $g$ are
\emph{\color{blue}orthogonal} on an interval $[a,b]$ if
\[
\int_a^bf(x)g(x)\,dx=0.
\]
More generally, we say that the functions $\phi_1$, $\phi_2$, \dots,
$\phi_n$, \dots (finitely or infinitely many) are orthogonal
on $[a,b]$  if
\[
\int_a^b\phi_i(x)\phi_j(x)\,dx=0\quad\text{whenever}\quad
i\ne j.
\]
The  importance of orthogonality will become clear when we study
Fourier
 series in the next two sections.

\begin{example}\label{example:11.1.4}
Show that the eigenfunctions
\begin{equation} \label{eq:11.1.11}
1,\, \cos\frac{\pi x}{L},\, \sin\frac{\pi x}{L},
\, \cos\frac{2\pi x}{L}, \, \sin\frac{2\pi x}{L},\dots,
\cos\frac{n\pi x}{L}, \, \sin\frac{n\pi x}{L},\dots
\end{equation}
of Problem 5
are orthogonal on $[-L,L]$.
\end{example}


\begin{solution}
We must show that
\begin{equation} \label{eq:11.1.12}
\int_{-L}^L f(x)g(x)\,dx=0
\end{equation}
whenever $f$ and $g$ are distinct functions from \eqref{eq:11.1.11}.
If $r$ is any nonzero integer, then
\begin{equation} \label{eq:11.1.13}
\int_{-L}^L\cos\frac{r\pi x}{L}\,dx
=\frac{L}{r\pi}\sin\frac{r\pi x}{L}\bigg|_{-L}^L=0.
\end{equation}
and
\[
\int_{-L}^L\sin\frac{r\pi x}{L}\,dx
=-\frac{L}{r\pi}\cos\frac{r\pi x}{L}\bigg|_{-L}^L=0.
\]
Therefore \eqref{eq:11.1.12} holds  if $f\equiv1$
and $g$ is any other function in \eqref{eq:11.1.11}.

If $f(x)=\cos m\pi x/L$ and $g(x)=\cos n\pi x/L$ where $m$
and $n$ are distinct positive integers, then
\begin{equation} \label{eq:11.1.14}
\int_{-L}^L f(x)g(x)\,dx=\int_{-L}^L\cos\frac{m\pi x}{L}
\cos\frac{n\pi x}{L}\,dx.
\end{equation}
To evaluate this integral, we use the identity
\[
\cos A\cos B=\frac{1}{2}[\cos(A-B)+\cos(A+B)]
\]
with $A=m\pi x/L$ and $B=n\pi x/L$.  Then \eqref{eq:11.1.14} becomes
\[
\int_{-L}^L
f(x)g(x)\,dx=\frac{1}{2}\left[\int_{-L}^L\cos\frac{(m-n)\pi x}{L}\,dx
+\int_{-L}^L\cos\frac{(m+n)\pi x}{L}\,dx\right].
\]
Since $m-n$ and $m+n$ are both nonzero integers, \eqref{eq:11.1.13}
implies that the  integrals on the right are both zero. Therefore
 \eqref{eq:11.1.12} is true  in this case.


If $f(x)=\sin m\pi x/L$ and $g(x)=\sin n\pi x/L$ where $m$
and $n$ are distinct positive integers, then
\begin{equation} \label{eq:11.1.15}
\int_{-L}^L f(x)g(x)\,dx=\int_{-L}^L\sin\frac{m\pi x}{L}
\sin\frac{n\pi x}{L}\,dx.
\end{equation}
To evaluate this integral, we use the  identity
\[
\sin A\sin B=\frac{1}{2}[\cos(A-B)-\cos(A+B)]
\]
with $A=m\pi x/L$ and $B=n\pi x/L$.  Then \eqref{eq:11.1.15} becomes
\[
\int_{-L}^L
f(x)g(x)\,dx=\frac{1}{2}\left[\int_{-L}^L\cos\frac{(m-n)\pi x}{L}\,dx
-\int_{-L}^L\cos\frac{(m+n)\pi x}{L}\,dx\right]=0.
\]


If $f(x)=\sin m\pi x/L$ and $g(x)=\cos n\pi x/L$ where $m$
and $n$ are   positive integers (not necessarily distinct),
then
\[
\int_{-L}^L f(x)g(x)\,dx=\int_{-L}^L\sin\frac{m\pi x}{L}
\cos\frac{n\pi x}{L}\,dx=0
\]
because the integrand is an odd function and the limits are symmetric
about $x=0$.
\end{solution}

\section{Fourier Expansions I}\label{sec:fourier_i}

In \zcref{example:11.1.4} and
{Exercises 11.1.\zcref[noname]{exer:11.1.4}}--{11.1.\zcref[noname]{exer:11.1.22}} we saw
that the
  eigenfunctions of Problem~5 are orthogonal on $[-L,L]$ and the
eigenfunctions of Problems~1--4 are orthogonal on $[0,L]$.
In this section and the next we introduce some series expansions
in terms of these eigenfunctions.
We'll use these expansions  to solve partial differential equations in
\zcref{ch:fourier_pde}.


\begin{theorem}\label{thmtype:11.2.1}
Suppose the functions  $\phi_1$, $\phi_2$,
$\phi_3$, \dots, are orthogonal on $[a,b]$ and
\begin{equation} \label{eq:11.2.1}
\int_a^b\phi_n^2(x)\,dx\ne0,\quad n=1,2,3,\dots.
\end{equation}
Let $c_1$, $c_2$, $c_3$,\dots  be constants such that the partial sums
$f_N(x)=\sum_{m=1}^N c_m\phi_m(x)$ satisfy the inequalities
\[
|f_N(x)|\le M,\quad a\le x\le b,\quad N=1,2,3,\dots
\]
for some constant $M<\infty$. Suppose  also that
the series
\begin{equation} \label{eq:11.2.2}
f(x)=\sum_{m=1}^\infty c_m\phi_m(x)
\end{equation}
converges and is integrable on $[a,b]$.
Then
\begin{equation} \label{eq:11.2.3}
c_n=\frac{\dst\int_a^bf(x)\phi_n(x)\,dx}{\dst\int_a^b\phi_n^2(x)\,dx},\quad
n=1,2,3,\dots.
\end{equation}
\end{theorem}

\begin{proof}
Multiplying \eqref{eq:11.2.2} by $\phi_n$ and
integrating yields
\begin{equation} \label{eq:11.2.4}
\int_a^b f(x)\phi_n(x)\,dx=\int_a^b
\phi_n(x)\left(\sum_{m=1}^\infty c_m\phi_m(x)\right)\,dx.
\end{equation}
It can be shown that the
 boundedness of the partial sums
$\{f_N\}_{N=1}^\infty$ and the integrability of $f$
allow us to interchange the operations of integration and summation
on the right of \eqref{eq:11.2.4}, and rewrite \eqref{eq:11.2.4} as
\begin{equation} \label{eq:11.2.5}
\int_a^b f(x)\phi_n(x)\,dx=\sum_{m=1}^\infty c_m
\int_a^b\phi_n(x) \phi_m(x)\,dx.
\end{equation}
(This isn't  easy to prove.) Since
\[
\int_a^b\phi_n(x)\phi_m(x)\,dx=0\quad\text{if }m\ne n,
\]
\eqref{eq:11.2.5} reduces to
\[
\int_a^bf(x) \phi_n(x)\,dx=c_n\int_a^b\phi_n^2(x)\,dx.
\]
Now  \eqref{eq:11.2.1} implies \eqref{eq:11.2.3}.
\end{proof}

\zcref[S]{thmtype:11.2.1} motivates the next definition.

\begin{definition}\label{thmtype:11.2.2}
Suppose
$\phi_1$, $\phi_2$, \dots, $\phi_n$,\dots are
orthogonal on $[a,b]$ and
$\int_a^b\phi_n^2(x)\,dx\ne0$, $n=1$, $2$, $3$, \dots. Let
$f$ be integrable on $[a,b]$, and define
\begin{equation} \label{eq:11.2.6}
c_n=\frac{\dst\int_a^bf(x)\phi_n(x)\,dx}{\dst\int_a^b\phi_n^2(x)\,dx},\quad
n=1,2,3,\dots.
\end{equation}
Then the infinite series
$\sum_{n=1}^\infty c_n\phi_n(x)$ is called the
\emph{\color{blue}Fourier expansion of $f$ in terms of the orthogonal set}
$\{\phi_n\}_{n=1}^\infty$, and
$c_1$, $c_2$, \dots, $c_n$, \dots
are called the \emph{\color{blue}Fourier coefficients of $f$ with respect to}
$\{\phi_n\}_{n=1}^\infty$. We indicate the relationship between $f$
and its Fourier
 expansion by
\begin{equation} \label{eq:11.2.7}
f(x)\sim\sum_{n=1}^\infty c_n\phi_n(x),\quad  a\le x\le b.
\end{equation}
\end{definition}

You may wonder why we don't write
\[
f(x)=\sum_{n=1}^\infty c_n\phi_n(x),\quad  a\le x\le b,
\]
rather than \eqref{eq:11.2.7}.
Unfortunately, this isn't always true. The series on the right
may diverge for some or all values
of $x$ in $[a,b]$, or it may converge to $f(x)$ for some values of
$x$ and not for others.  So, for now,  we'll just think of the series
as being associated with $f$ because of the definition
of the coefficients $\{c_n\}$, and we'll indicate
this association informally as in \eqref{eq:11.2.7}.


\boxit{Fourier Series}

We'll now study Fourier expansions in terms
 of the eigenfunctions
\[
1,\, \cos\frac{\pi x}{L},\, \sin\frac{\pi x}{L},
\, \cos\frac{2\pi x}{L}, \, \sin\frac{2\pi x}{L},\dots,
\cos\frac{n\pi x}{L}, \, \sin\frac{n\pi x}{L},\dots.
\]
of Problem~5.
 If $f$ is integrable on $[-L,L]$,  its
Fourier expansion in terms of these functions is called the
\emph{\color{blue}Fourier series of $f$ on $[-L,L]$}.  Since
\begin{gather*}
\int_{-L}^L 1^2\,dx=2L,\\
\int_{-L}^L\cos^2\frac{n\pi x}{L}\,dx =
\frac{1}{2}\int_{-L}^L\left(1+\cos\frac{2n\pi x}{L}\right)\,dx=
\frac{1}{2}\left(x+\frac{L}{2n\pi}\sin\frac{2n\pi x}{L}\right)
\lims{-L}L=L,
\end{gather*}
and
\[
\int_{-L}^L\sin^2\frac{n\pi x}{L}\,dx =
\frac{1}{2}\int_{-L}^L\left(1-\cos\frac{2n\pi x}{L}\right)\,dx=
\frac{1}{2}\left(x-\frac{L}{2n\pi}\sin\frac{2n\pi x}{L}\right),
\lims{-L}L=L,
\]
we see from \eqref{eq:11.2.6} that the Fourier series of $f$ on $[-L,L]$ is
\[
 a_0+\sum_{n=1}^\infty \left(a_n\cos\frac{n\pi x}{L}+b_n\sin\frac{n\pi x}{L}\right),
\]
where
\begin{gather*}
a_0=\frac{1}{2L}\int_{-L}^L f(x)\,dx,\\
a_n= \frac{1}{L}\int_{-L}^L f(x)\cos\frac{n\pi
x}{L}\,dx,\quad\text{and}\quad
b_n=\frac{1}{L}\int_{-L}^L f(x)\sin\frac{n\pi x}{L}\,dx,\,n\ge1.
\end{gather*}

Note that
$a_0$ is the average value of $f$ on $[-L,L]$, while $a_n$ and $b_n$
(for $n\ge1$) are twice the average values of
\[
f(x)\cos\frac{n\pi x}{L}\quad\text{and}\quad f(x)\sin\frac{n\pi x}{L}
\]
on $[-L,L]$, respectively.


\boxit{Convergence of Fourier Series}

The  question of convergence of Fourier series for arbitrary
integrable functions is beyond the scope of this book. However, we can
state a theorem that settles this question for most functions that
arise in applications.

\begin{definition}\label{thmtype:11.2.3}
A function  $f$ is said to be
\emph{\color{blue}piecewise smooth} on $[a,b]$ if:
\begin{alist}
\item % (a)
$f$ has at most finitely many points of discontinuity in $(a,b)$;
\item % (b)
$f'$ exists and is continuous except possibly at finitely many points
in $(a,b)$;
\item % (c)
$f(x_0+)=\lim_{x\to x_0+}f(x)$ and
$f'(x_0+)=\lim_{x\to x_0+}f'(x)$ exist if $a\le x_0<b$;
\item % (d)
$f(x_0-)=\lim_{x\to x_0-}f(x)$ and
$f'(x_0-)=\lim_{x\to x_0-}f'(x)$ exist if  $a< x_0\le b$.
\end{alist}
\end{definition}

Since $f$ and $f'$ are required to be continuous at all but
finitely many points in $[a,b]$,
 $f(x_0+)=f(x_0-)$
and $f'(x_0+)=f'(x_0-)$ for all but finitely many
values of $x_0$ in $(a,b)$. Recall from \zcref{sec:laplace_intro} that
$f$ is said to have a \emph{\color{blue}jump discontinuity} at $x_0$
if $f(x_0+)\ne f(x_0-)$.

The next theorem gives sufficient conditions for convergence of a
Fourier series. The  proof is beyond the scope of this book.


\begin{theorem}\label{thmtype:11.2.4}
If $f$ is piecewise smooth on $[-L,L]$, then the Fourier series
\begin{equation} \label{eq:11.2.8}
F(x)=a_0+\sum_{n=1}^\infty\left(a_n\cos\frac{n\pi x}{L}+b_n\sin\frac{n\pi
x}{L}\right)
\end{equation}
of $f$
on $[-L,L]$ converges for all $x$ in $[-L,L]$; moreover,
\[
F(x)=
\begin{cases}
f(x)&\text{if $-L<x<L$ and $f$ is continuous at $x$}\\
\dfrac{f(x-)+f(x+)}{2}&\text{if $-L<x<L$ and $f$ is discontinuous at
$x$}\\
\dfrac{f(-L+)+f(L-)}{2}&\text{if $x=L$ or $x=-L$.}
\end{cases}
\]
\end{theorem}


Since $f(x+)=f(x-)$ if $f$ is continuous at $x$, we can also say that
\[
F(x)=
\begin{cases}
\dfrac{f(x+)+f(x-)}{2}&\text{if }-L<x<L,\\
\dfrac{f(L-)+f(-L+)}{2}&\text{if }x=\pm L.
\end{cases}
\]



Note that $F$ is itself  piecewise smooth on $[-L,L]$, and
 $F(x)=f(x)$ at all points in  the open interval
$(-L,L)$ where $f$ is continuous.
 Since the series in \eqref{eq:11.2.8}
converges to $F(x)$ for all $x$ in $[-L,L]$, you may be tempted to
infer
 that the error
\[
E_N(x)=
\left|F(x)-a_0-\sum_{n=1}^N\left(a_n\cos\frac{n\pi x}{L}+b_n\sin\frac{n\pi
x}{L}\right)\right|
\]
can be made as small as we please  for all $x$ in $[-L,L]$
by choosing $N$ sufficiently large. However, this isn't  true if
$f$ has a  discontinuity  somewhere in $(-L,L)$, or if $f(-L+)\ne
f(L-)$.  Here's  the situation in this case.

\begin{remark*}
If $f$ has a jump discontinuity at a point $\alpha$ in $(-L,L)$,  there
will be sequences
of points $\{u_N\}$ and $\{v_N\}$ in $(-L,\alpha)$ and $(\alpha,L)$,
respectively, such that
\[
\lim_{N\to\infty}u_N=
\lim_{N\to\infty}v_N=\alpha
\]
and
\[
E_N(u_N)\approx .09|f(\alpha-)-f(\alpha+)|\quad\text{and}\quad
E_N(v_N)\approx .09|f(\alpha-)-f(\alpha+)|.
\]
Thus, the maximum value of the error $E_N(x)$ near $\alpha$ does not
approach zero as $N\to\infty$, but just occurs closer and closer to
(and on both sides of) $\alpha$, and is essentially independent
of
$N$.

If $f(-L+)\ne f(L-)$, then there will be sequences of points
 $\{u_N\}$ and $\{v_N\}$ in $(-L,L)$
such that
\begin{gather*}
\lim_{N\to\infty}u_N=-L,\quad
\lim_{N\to\infty}v_N=L,\\
E_N(u_N)\approx .09|f(-L+)-f(L-)|\quad\text{and}\quad
E_N(v_N)\approx .09|f(-L+)-f(L-)|.
\end{gather*}
\end{remark*}

This  is the
\personHref{Gibbs}{Gibbs phenomenon}. Having been alerted to
it, you may see it in \zcref[range]{figure:11.2.2,figure:11.2.4},
 below;
however, we'll give a  specific example at the end of this section.

\begin{example}\label{example:11.2.1}
Find the Fourier series of the piecewise smooth function
\[
f(x)=
\begin{cases}
-x,&-2< x<0,\\
\frac{1}{2},&\phantom{-}0<x<2
\end{cases}
\]
on $[-2,2]$ (\zcref{figure:11.2.1}). Determine the sum of
the Fourier series  for $-2\le x\le 2$.
\end{example}

  \centering\color{blue}
  \includegraphics[alt={A graph of the function in Example \ref*{example:11.2.1}},
  width=5.67in,height=3.3in,keepaspectratio]{fig110201}
\caption{}
  \label{figure:11.2.1}
\end{figure}


\begin{solution}
Note that we haven't bothered to define $f(-2)$, $f(0)$, and
$f(2)$.
 No matter
how they may be defined, $f$ is piecewise smooth on $[-2,2]$, and
the coefficients in the Fourier series
\[
F(x)=a_0+\sum_{n=1}^\infty\left(a_n\cos\frac{n\pi
x}{2}+b_n\sin\frac{n\pi x}{2}\right)
\]
are not affected by them. In any case,
\zcref{thmtype:11.2.4} implies that $F(x)=f(x)$ in $(-2,0)$
and $(0,2)$, where $f$ is continuous, while
\[
F(-2)=F(2)=\frac{f(-2+)+f(2-)}{2}=
\frac{1}{2}\left(2+\frac{1}{2}\right)=\frac{5}{4}
\]
and
\[
F(0)=\frac{f(0-)+f(0+)}{2}=\frac{1}{2}\left(0+\frac{1}{2}\right)=\frac{1}{4}.
\]
To summarize,
\[
F(x)=
\begin{cases}
\frac{5}{4},&\phantom{-}x=-2\\
-x,&-2<x<0,\\
 \frac{1}{4},&\phantom{-}x=0,\\
\frac{1}{2},&\phantom{-}0<x<2,\\
\frac{5}{4},&\phantom{-}x=2.
\end{cases}
\]

We compute the Fourier coefficients as follows:
\[
a_0=\frac{1}{4}\int_{-2}^2f(x)\,dx=\frac{1}{4}\left[\int_{-2}^0(-x)\,dx
+\int_0^2\frac{1}{2}\,dx\right]
=\frac{3}{4}.
\]
If $n\ge1$, then
\begin{align*}
a_n&=\frac{1}{2}\int_{-2}^2f(x)\cos\frac{n\pi x}{2}\,dx
=\frac{1}{2}\left[\int_{-2}^0(-x)\cos\frac{n\pi x}{2}\,dx
+\int_0^2\frac{1}{2}\cos\frac{n\pi x}{2}\,dx\right]\\
&=\frac{2}{n^2\pi^2}(\cos n\pi-1),
\end{align*}
and
\begin{align*}
b_n&=\frac{1}{2}\int_{-2}^2f(x)\sin\frac{n\pi x}{2}\,dx
=\frac{1}{2}\left[\int_{-2}^0(-x)\sin\frac{n\pi x}{2}\,dx
+\int_0^2\frac{1}{2}\sin\frac{n\pi x}{2}\,dx\right]\\
&=\frac{1}{2n\pi}(1+3\cos n\pi).
\end{align*}
Therefore
\[
F(x)=\frac{3}{4}+\frac{2}{\pi^2}\sum_{n=1}^\infty\frac{\cos n\pi-1}{n^2}\cos\frac{n\pi x}{2}+\frac{1}{2\pi}\sum_{n=1}^\infty\frac{1+3\cos
n\pi}{n}\sin\frac{n\pi x}{2}.
\]
\zcref[S]{figure:11.2.2} shows how the partial sum
\[
F_m(x)=\frac{3}{4}+\frac{2}{\pi^2}\sum_{n=1}^m\frac{\cos n\pi-1}{n^2}\cos\frac{n\pi x}{2}+\frac{1}{2\pi}\sum_{n=1}^m\frac{1+3\cos
n\pi}{n}\sin\frac{n\pi x}{2}
\]
approximates $f(x)$ for $m=5$ (dotted curve), $m=10$  (dashed curve),
and $m=15$ (solid curve).
\end{solution}

\boxit{Even and Odd Functions}

Computing the Fourier coefficients of a  function  $f$
 can be tedious;     however,  the computation
can often be simplified by exploiting symmetries
in
$f$ or  some of its terms. To focus on this,
we recall some concepts that you studied in calculus.
Let $u$ and $v$ be defined on $[-L,L]$ and suppose
that
\[
u(-x)=u(x)\quad\text{and}\quad v(-x)=-v(x),\quad -L\le x\le L.
\]
Then we say that $u$ is an \emph{\color{blue}even} function and $v$  is an
\emph{\color{blue}odd function}. Note that:
\begin{itemize}
\item
The product of two even functions is even.
\item
The product of two odd functions is even.
\item
The product of an even  function and an odd function  is odd.
\end{itemize}

\begin{example}\label{example:11.2.2}
The functions $u(x)=\cos \omega x$ and $u(x)=x^2$  are even, while
$v(x)=\sin \omega x$ and $v(x)=x^3$ are odd. The function $w(x)=e^x$
is neither even nor odd.
\end{example}



You learned parts (a) and (b) of the  next theorem in
calculus, and the other parts follow from them
(\zcref{exer:11.2.1}).


\begin{theorem}\label{thmtype:11.2.5}
Suppose $u$ is even and $v$ is odd on $[-L,L]$. Then:
\begin{enumext*}[label=\alph*,wrap-label={(#1)},columns=2,labelwidth=2em]
\item $\dst\int_{-L}^L u(x)\,dx=2\int_0^Lu(x) \,dx$,
\item $\dst\int_{-L}^L v(x)\,dx=0$,
\item(2) $\dst\int_{-L}^L u(x)\cos\frac{n\pi x}{L}\,dx
=2\int_0^Lu(x)\cos\frac{n\pi x}{L} \,dx$,
\item(2) $\dst\int_{-L}^L v(x)\sin\frac{n\pi x}{L}\,dx=
2\int_0^L v(x)\sin\frac{n\pi x}{L}\,dx$,
\item $\dst\int_{-L}^L u(x)\sin\frac{n\pi x}{L}\,dx=0$\qquad and
\item $\dst\int_{-L}^L v(x)\cos\frac{n\pi x}{L}\,dx=0$.
\end{enumext*}
\end{theorem}

\begin{example}\label{example:11.2.3}
Find the Fourier series of $f(x)=x^2-x$ on $[-2,2]$, and determine
its sum for $-2\le x\le 2$.
\end{example}

\begin{solution}
Since $L=2$,
\[
F(x)=a_0+\sum_{n=1}^\infty\left(a_n\cos\frac{n\pi
x}{2}+b_n\sin\frac{n\pi x}{2}\right)
\]
where
\begin{equation} \label{eq:11.2.9}
a_0=\frac{1}{4}\int_{-2}^2(x^2-x)\,dx,
\end{equation}
\begin{equation} \label{eq:11.2.10}
a_n=\frac{1}{2}\int_{-2}^2(x^2-x)\cos\frac{n\pi x}{2}\,dx,\quad
n=1,2,3,\dots,
\end{equation}
and
\begin{equation} \label{eq:11.2.11}
b_n=\frac{1}{2}\int_{-2}^2(x^2-x)\sin\frac{n\pi x}{2}\,dx,\quad
n=1,2,3,\dots.
\end{equation}
We simplify the evaluation of these integrals by using
\zcref{thmtype:11.2.5}   with $u(x)=x^2$ and $v(x)=x$;
thus, from \eqref{eq:11.2.9},
\[
a_0=\frac{1}{2}\int_0^2x^2\,dx=\frac{x^3}{6}\lims02=\frac{4}{3}.
\]
From \eqref{eq:11.2.10},
\begin{align*}
a_n&=\int_0^2x^2\cos\frac{n\pi x}{2}\,dx=
\frac{2}{n\pi}\left[x^2\sin\frac{n\pi x}{2}\lims02-
2\int_0^2x\sin\frac{n\pi x}{2}\,dx\right]\\
&=\frac{8}{n^2\pi^2}\left[x\cos\frac{n\pi x}{2}\lims02-
\int_0^2\cos\frac{n\pi x}{2}\,dx\right]\\
&=\frac{8}{n^2\pi^2}\left[2\cos n\pi-\frac{2}{n\pi}\sin\frac{n\pi x}{2}\lims02\right]
=(-1)^n\frac{16}{n^2\pi^2}.
\end{align*}
From \eqref{eq:11.2.11},
\begin{align*}
b_n&=-\int_0^2x\sin\frac{n\pi x}{2}\,dx
=\frac{2}{n\pi}\left[x\cos\frac{n\pi x}{2}\lims02-
\int_0^2\cos\frac{n\pi x}{2}\,dx\right]\\
&=\frac{2}{n\pi}\left[2\cos n\pi-\frac{2}{n\pi}\sin\frac{n\pi x}{2}\lims02\right]=(-1)^n\frac{4}{n\pi}.
\end{align*}
Therefore
\[
F(x)=\frac{4}{3}+\frac{16}{\pi^2}\sum_{n=1}^\infty\frac{(-1)^n}{n^2}
\cos\frac{n\pi x}{2}+\frac{4}{\pi}\sum_{n=1}^\infty
\frac{(-1)^n}{n}\sin\frac{n\pi x}{2}.
\]
\zcref[S]{thmtype:11.2.4} implies that
\[
F(x)=
\begin{cases}
4,&\phantom{-}x=-2,\\
x^2-x,&-2<x<2,\\
4,&\phantom{-}x=2.
\end{cases}
\]
\zcref[S]{figure:11.2.3} shows how the partial sum
\[
F_m(x)=\frac{4}{3}+\frac{16}{\pi^2}\sum_{n=1}^m\frac{(-1)^n}{n^2}
\cos\frac{n\pi x}{2}+\frac{4}{\pi}\sum_{n=1}^m
\frac{(-1)^n}{n}\sin\frac{n\pi x}{2}
\]
approximates $f(x)$ for $m=5$ (dotted curve), $m=10$  (dashed curve),
and $m=15$ (solid curve).
\end{solution}

the next theorem follows.


\begin{theorem}\label{thmtype:11.2.6}
Suppose   $f$ is integrable on $[-L,L]$.
\begin{alist}
\item % (a)
If $f$ is even,  the Fourier series of $f$ on $[-L,L]$ is
\[
F(x)=a_0+\sum_{n=1}^\infty a_n\cos\frac{n\pi x}{L},
\]
where
\[
a_0=\frac{1}{L}\int_0^Lf(x) \,dx \quad\text{and}\quad
a_n=\frac{2}{L}\int_0^L f(x)\cos\frac{n\pi x}{L}\,dx,\quad n\ge1.
\]
\item % (b)
If $f$ is odd,  the Fourier series of $f$ on $[-L,L]$ is
\[
F(x)= \sum_{n=1}^\infty b_n \sin\frac{n\pi x}{L},
\]
where
\[
b_n=\frac{2}{L}\int_0^L f(x)\sin\frac{n\pi x}{L}\,dx.
\]
\end{alist}
\end{theorem}




\begin{example}\label{example:11.2.4}
Find the Fourier series of $f(x)=x$ on $[-\pi,\pi]$, and determine
its sum for $-\pi\le x\le \pi$.
\end{example}

\begin{solution}
Since $f$ is odd and $L=\pi$,
\[
F(x)=\sum_{n=1}^\infty b_n\sin nx
\]
where
\begin{align*}
b_n&=\frac{2}{\pi}\int_0^\pi x\sin nx\,dx=-\frac{2}{n\pi}\left[x\cos
nx\lims0\pi-\int_0^\pi\cos nx\,dx\right]\\
&=-\frac{2}{n}\cos n\pi+\frac{2}{n^2\pi}\sin nx\lims0\pi=(-1)^{n+1}
\frac{2}{n}.
\end{align*}
Therefore
\[
F(x)=-2\sum_{n=1}^\infty\frac{(-1)^n}{n}\sin nx.
\]
\zcref[S]{thmtype:11.2.4} implies that
\[
F(x)=
\begin{cases}
0,&\phantom{-}x=-\pi,\\
x,&-\pi<x<\pi,\\
0,&\phantom{-}x=\pi.
\end{cases}
\]
\zcref[S]{figure:11.2.4} shows how the partial sum
\[
F_m(x)=-2\sum_{n=1}^m\frac{(-1)^n}{n}\sin nx
\]
approximates $f(x)$ for $m=5$ (dotted curve), $m=10$  (dashed curve),
and $m=15$ (solid curve).
\end{solution}

\begin{example}\label{example:11.2.5}
Find the Fourier series of $f(x)=|x|$ on $[-\pi,\pi]$ and
determine its sum for\\ $-\pi\le x\le\pi$.
\end{example}

\begin{solution}
Since $f$ is even and $L=\pi$,
\[
F(x)=a_0+\sum_{n=1}^\infty a_n\cos nx.
\]
Since $f(x)=x$ if $x\ge0$,
\[
a_0=\frac{1}{\pi}\int_0^\pi x\,dx=\frac{x^2}{2\pi}\lims0\pi=\frac{\pi}{2}
\]
and, if $n\ge1$,
\begin{align*}
a_n&=\frac{2}{\pi}\int_0^\pi x\cos nx\,dx=\frac{2}{n\pi}\left[x\sin
nx\lims0\pi-\int_0^\pi\sin nx\,dx\right]\\
&=\frac{2}{n^2\pi}\cos
nx\lims0\pi=\frac{2}{n^2\pi}(\cos n\pi-1)=\frac{2}{n^2\pi}[(-1)^n-1].
\end{align*}
Therefore
\begin{equation} \label{eq:11.2.12}
F(x)=\frac{\pi}{2}+\frac{2}{\pi}\sum_{n=0}\frac{(-1)^n-1}{n^2}\cos nx.
\end{equation}
However, since
\[
(-1)^n-1=
\begin{cases}
0&\text{ if }n=2m,\\
-2&\text{ if }n=2m+1,
\end{cases}
\]
the terms in  \eqref{eq:11.2.12} for which $n=2m$ are all zeros.
Therefore we only to include the terms for which $n=2m+1$;
that is, we can rewrite \eqref{eq:11.2.12} as
\[
F(x)=\frac{\pi}{2}-\frac{4}{\pi}\sum_{m=0}^\infty \frac{1}{(2m+1)^2}
\cos(2m+1)x.
\]
However, since the name of the index of summation doesn't matter,
we prefer to replace $m$ by $n$, and write
\[
F(x)=\frac{\pi}{2}-\frac{4}{\pi}\sum_{n=0}^\infty \frac{1}{(2n+1)^2}
\cos(2n+1)x.
\]
Since $|x|$ is continuous for all $x$ and $|-\pi|=|\pi|$,
\zcref{thmtype:11.2.4} implies that $F(x)=|x|$ for
all $x$ in $[-\pi,\pi]$.
\end{solution}

\begin{example}\label{example:11.2.6}
Find the Fourier series of $f(x)=x(x^2-L^2)$  on $[-L,L]$,
and determine its sum for $-L\le x\le L$.
\end{example}

\begin{solution}
Since $f$ is odd,
\[
F(x)=\sum_{n=1}^\infty b_n\sin\frac{n\pi x}{L},
\]
where
\begin{align*}
b_n&=\frac{2}{L}\int_0^Lx(x^2-L^2)\sin\frac{n\pi x}{L}\,dx\\
&=-\frac{2}{n\pi}\left[x(x^2-L^2)\cos\frac{n\pi x}{L}\lims0L-
\int_0^L
(3x^2-L^2)\cos\frac{n\pi x}{L}\,dx\right]\\
&=\frac{2L}{n^2\pi^2}\left[(3x^2-L^2)\sin\frac{n\pi x}{L}\lims0L-6
\int_0^Lx\sin\frac{n\pi x}{L}\,dx\right]\\
&=\frac{12L^2}{n^3\pi^3}\left[x\cos\frac{n\pi x}{L}\lims0L-
\int_0^L\cos\frac{n\pi x}{L}\,dx\right]
=(-1)^n\frac{12L^3}{n^3\pi^3}.
\end{align*}
Therefore
\[
F(x)=\frac{12L^3}{\pi^3}\sum_{n=1}^\infty\frac{(-1)^n}{n^3}\sin\frac{n\pi
x}{L}.
\]
\zcref[S]{thmtype:11.2.4} implies that $F(x)=x(x^2-L^2)$ for all $x$
in $[-L,L]$.
\end{solution}

\begin{example}[Gibbs Phenomenon]\label{example:11.2.7}
The Fourier series of
\[
f(x)=\begin{cases}
0,&-1<x<-\frac{1}{2},\\
1,&-\frac{1}{2}<x<\frac{1}{2},\\
0,&\phantom{-}\frac{1}{2}<x<1
\end{cases}
\]
on $[-1,1]$ is
\[
F(x)=\frac{1}{2}+\frac{2}{\pi}\sum_{n=1}^\infty
\frac{(-1)^{n-1}}{2n-1}\cos(2n-1)\pi x.
\]
(Verify.) According to \zcref{thmtype:11.2.4},
\[
F(x)=\begin{cases}
0,&-1\le x<-\frac{1}{2},\\
\frac{1}{2},& x=-\frac{1}{2},\\
1,&-\frac{1}{2}<x<\frac{1}{2},\\
\frac{1}{2},& x=\frac{1}{2},\\
0,&\frac{1}{2}<x\le 1;
\end{cases}
\]
thus, $F$ (as well as $f$) has unit jump discontinuities
at $x=\pm\frac{1}{2}$.
\zcref[S,range]{figure:11.2.5,figure:11.2.7} show the graphs
of $y=f(x)$ and

approximates $F$ (and therefore $f$) well on larger intervals as $N$
increases, the maximum absolute values of the errors

discontinuities $x=\pm\frac{1}{2}$ as $N$ increases.
\end{example}

\boxit{Fourier Cosine Series}

From {Exercise 11.1.\zcref[noname]{exer:11.1.20}}, the eigenfunctions
\[
1,\, \cos\frac{\pi x}{L},
\, \cos\frac{2\pi x}{L},\dots,
 \, \cos\frac{n\pi x}{L},\dots
\]
 of the boundary value problem
\begin{equation} \label{eq:11.3.1}
y''+\lambda y=0,\quad y'(0)=0,\quad y'(L)=0
\end{equation}
(Problem~2)  are orthogonal on $[0,L]$.
 If $f $ is integrable on $[0,L]$ then the
Fourier expansion of $f$ in terms of these functions is called the
\emph{\color{blue}Fourier cosine series of $f$ on $[0,L]$}. This series is
\[
a_0+\sum_{n=1}^\infty a_n\cos\frac{n\pi x}{L},
\]
where
\[
a_0=\frac{\dst\int_0^Lf(x)\,dx}{\dst\int_0^L\,dx}=\frac{1}{L}\int_0^Lf(x)\,dx
\]
and
\[
a_n=\frac{\dst\int_0^Lf(x)\cos\frac{n\pi x}{L}\,dx}{\dst\int_0^L
\cos^2\frac{n\pi x}{L}\,dx}=\frac{2}{L}\int_0^Lf(x)\cos\frac{n\pi x}{L}\,dx,\quad n=1,2,3,\dots.
\]
Comparing this definition with
\zcref{exer:11.2.6}(a) shows that the Fourier cosine
series of $f$ on $[0,L]$ is  the Fourier series of
 the  function
\[
f_1(x)=
\begin{cases}
f(-x),&-L<x<0,\\f(x),&\phantom{-}0\le x\le L,
\end{cases}
\]
obtained by extending  $f$ over $[-L,L]$ as an even
function (\zcref{figure:11.3.1}).

\begin{figure}
  \centering\color{blue}
  \includegraphics[alt={A function for 0<x<L.  The portion for -L<x<0 is dashed, and is a mirror image across the y axis of the portion for 0<x<L.},
  width=5.67in,height=3.3in,keepaspectratio]{fig110301}
  \caption{}
  \label{figure:11.3.1}
\end{figure}

Applying
\zcref{thmtype:11.2.4} to $f_1$ yields the next
theorem.

\begin{theorem}\label{thmtype:11.3.1}
If $f$ is piecewise smooth on $[0,L]$, then the Fourier cosine series
\[
C(x)=a_0+\sum_{n=1}^\infty a_n\cos\frac{n\pi x}{L}
\]
of $f$
on $[0,L]$,
with
\[
a_0=\frac{1}{L}\int_0^Lf(x)\,dx
\quad\text{and}\quad
a_n=\frac{2}{L}\int_0^Lf(x)\cos\frac{n\pi x}{L}\,dx,\quad n=1,2,3,\dots,
\]
 converges for all $x$ in $[0,L]$; moreover,
\[
C(x)=
\begin{cases}
f(0+)&\text{ if }x=0  \\
f(x)&\text{ if $0<x<L$ and $f$ is continuous at $x  $}\\
\dfrac{f(x-)+f(x+)}{2}&\text{ if $0<x<L$ and $f$ is discontinuous at
$x  $}\\
f(L-)&\text{ if }x=L.
\end{cases}
\]
\end{theorem}

\begin{example}\label{example:11.3.1}
Find the Fourier cosine series of $f(x)=x$ on $[0,L]$.
\end{example}

\begin{solution}
 The coefficients are
\[
a_0=\frac{1}{L}\int_0^Lx\,dx=\frac{1}{L}\frac{x^2}{2}
\lims0L=\frac{L}{2}
\]
and, if $n\ge1$
\begin{align*}
a_n&=\frac{2}{L}\int_0^Lx\cos\frac{n\pi x}{L}\,dx
=\frac{2}{n\pi}\left[x\sin\frac{n\pi x}{L}\lims0L-
\int_0^L
\sin\frac{n\pi x}{L}\,dx\right]\\
&=-\frac{2}{n\pi}\int_0^L
\sin\frac{n\pi x}{L}\,dx
=\frac{2L}{n^2\pi^2}\cos\frac{n\pi x}{L}\lims0L
=\frac{2L}{n^2\pi^2}[(-1)^n-1]\\
&=\begin{cases}
\dst-\frac{4L}{(2m-1)^2\pi^2}&\text{ if  $n=2m-1$},\\
0&\text{ if $n=2m$}.
\end{cases}
\end{align*}
Therefore
\[
C(x)=\frac{L}{2}-\frac{4L}{\pi^2}\sum_{n=1}^\infty\frac{1}{(2n-1)^2}
\cos\frac{(2n-1)\pi x}{L}.
\]
\zcref[S]{thmtype:11.3.1} implies that
\[
C(x)=x,\quad 0\le x\le L.\qedhere
\]
\end{solution}


\boxit{Fourier Sine Series}

From Exercise 11.1.\zcref[noname]{exer:11.1.19}, the eigenfunctions
\[
\sin\frac{\pi x}{L},
\, \sin\frac{2\pi x}{L},\dots,
 \, \sin\frac{n\pi x}{L},\dots
\]
 of the boundary value problem
\[
y''+\lambda y=0,\quad y(0)=0,\quad y(L)=0
\]
(Problem~1)  are orthogonal on $[0,L]$.
 If $f$ is integrable on $[0,L]$ then the
Fourier expansion of $f$ in terms of these functions is called the
\emph{\color{blue}Fourier sine series of $f$ on $[0,L]$}. This series is
\[
\sum_{n=1}^\infty b_n\sin\frac{n\pi x}{L},
\]
where
\[
b_n=\frac{\dst\int_0^Lf(x)\sin\frac{n\pi x}{L}\,dx}{\dst\int_0^L
\sin^2\frac{n\pi x}{L}\,dx}=\frac{2}{L}\int_0^Lf(x)\sin\frac{n\pi x}{L}\,dx,\quad n=1,2,3,\dots.
\]
Comparing this definition with
\zcref{exer:11.2.6}(b) shows that the Fourier sine
series of $f$ on $[0,L]$ is the Fourier series of
 the  function
\[
f_2(x)=
\begin{cases}
-f(-x),&-L<x<0,\\f(x),&\phantom{-}0\le x\le L,
\end{cases}
\]
obtained by extending  $f$ over $[-L,L]$ as an odd
function  (\zcref{figure:11.3.2}).

  \centering\color{blue}
  \includegraphics[alt={A function for 0<x<L.  The portion for -L<x<0 is dashed, and is a mirror image across the origin of the portion for 0<x<L.},
  width=5.67in,height=3.3in,keepaspectratio]{fig110302}
  \caption{}
  \label{figure:11.3.2}
\end{figure}

Applying \zcref{thmtype:11.2.4} to $f_2$ yields the next theorem.

\begin{theorem}\label{thmtype:11.3.2}
If $f$ is piecewise smooth on $[0,L]$, then the Fourier sine series
\[
S(x)=\sum_{n=1}^\infty b_n\sin\frac{n\pi x}{L}
\]
of $f$ on $[0,L]$, with
\[
b_n=\frac{2}{L}\int_0^Lf(x)\sin\frac{n\pi x}{L}\,dx,
\]
 converges for all $x$ in $[0,L]$; moreover,
\[
S(x)=
\begin{cases}
0&\text{ if }x=0  \\
f(x)&\text{ if $0<x<L$ and $f$ is continuous at $x  $}\\
\dfrac{f(x-)+f(x+)}{2}&\text{ if $0<x<L$ and $f$ is discontinuous at
$x  $}\\
0&\text{ if }x=L.
\end{cases}
\]
\end{theorem}


\begin{example}\label{example:11.3.2}
Find the Fourier sine series of $f(x)=x$ on $[0,L]$.
\end{example}

\begin{solution}
 The coefficients are
\begin{align*}
b_n&=\frac{2}{L}\int_0^Lx\sin\frac{n\pi x}{L}\,dx
=-\frac{2}{n\pi}\left[x\cos\frac{n\pi x}{L}\lims0L-
\int_0^L
\cos\frac{n\pi x}{L}\,dx\right]\\
&= (-1)^{n+1}\frac{2L}{n\pi}+\frac{2L}{n^2\pi^2}\sin\frac{n\pi x}{L}\lims0L =(-1)^{n+1}\frac{2L}{n\pi}.
\end{align*}
Therefore
\[
S(x)=-\frac{2L}{\pi}\sum_{n=1}^\infty\frac{(-1)^n}{n}
\sin\frac{n\pi x}{L}.
\]
\zcref[S]{thmtype:11.3.2} implies that
\[
S(x)=
\begin{cases}
x,&0\le x< L,\\0,& x=L.
\end{cases}\qedhere
\]
\end{solution}


\boxit{Mixed Fourier Cosine Series}

From {Exercise 11.1.\zcref[noname]{exer:11.1.22}}, the eigenfunctions
\[
 \cos\frac{\pi x}{2L},
\, \cos\frac{3\pi x}{2L},\dots,
 \, \cos\frac{(2n-1)\pi x}{2L},\dots
\]
 of the boundary value problem
\begin{equation} \label{eq:11.3.2}
y''+\lambda y=0,\quad y'(0)=0,\quad y(L)=0
\end{equation}
(Problem~4) are orthogonal on $[0,L]$.
 If $f$ is integrable on $[0,L]$ then the
Fourier expansion of $f$ in terms of these functions  is
\[
\sum_{n=1}^\infty c_n\cos\frac{(2n-1)\pi x}{2L},
\]
where
\[
c_n=\frac{\dst\int_0^Lf(x)\cos\frac{(2n-1)\pi x}{2L}\,dx}{\dst\int_0^L
\cos^2\frac{(2n-1)\pi x}{L}\,dx}=\frac{2}{L}\int_0^Lf(x)\cos\frac{(2n-1)\pi
x}{2L}\,dx.
\]

We'll  call this expansion the \emph{\color{blue}mixed Fourier cosine
series} of $f$ on $[0,L]$, because the boundary conditions
of \eqref{eq:11.3.2} are ``mixed'' in that they require $y$
to be zero at
 one boundary point and $y'$ to be  zero at the other. By contrast,
the ``ordinary'' Fourier cosine series is associated with
\eqref{eq:11.3.1}, where the boundary conditions require that $y'$
be zero at both endpoints.

It can be shown (\zcref{exer:11.3.57}) that
 the mixed Fourier cosine series
of $f$ on $[0,L]$
is simply the restriction to  $[0,L]$ of the Fourier
cosine series  of
\[
f_3(x)=
\begin{cases}
f(x),&0\le x\le L,\\-f(2L-x),&L< x\le 2L
\end{cases}
\]
on $[0,2L]$ (\zcref{figure:11.3.3}).

\begin{figure}
  \centering\color{blue}
  \includegraphics[alt={A function for 0<x<L.  The portion for L<x<2L is dashed, and is a mirror image across the point (L,0) of the portion for 0<x<L.},
  width=5.67in,height=3.3in,keepaspectratio]{fig110303}
  \caption{}
  \label{figure:11.3.3}
\end{figure}

Applying  \zcref{thmtype:11.3.1}  with $f$ replaced by $f_3$ and $L$
replaced by $2L$ yields the next theorem.



\begin{theorem}\label{thmtype:11.3.3}
If  $f$ is piecewise smooth on $[0,L]$, then
the mixed Fourier cosine series
\[
C_M(x)=\sum_{n=1}^\infty c_n\cos\frac{(2n-1)\pi x}{2L}
\]
of $f$ on $[0,L]$,
with
\[
c_n=\frac{2}{L}\int_0^Lf(x)\cos\frac{(2n-1)\pi x}{2L}\,dx,
\]
 converges for all $x$ in $[0,L];    $ moreover,
\[
C_M(x)=
\begin{cases}
f(0+)&\text{ if $x=0$}  \\
f(x)&\text{ if $0<x<L$ and $f$ is continuous at $x$  }\\
\dfrac{f(x-)+f(x+)}{2}&\text{ if $0<x<L$ and $f$ is discontinuous at
$x$}\\
0&\text{ if $x=L$.}
\end{cases}
\]
\end{theorem}




\begin{example}\label{example:11.3.3}
Find  the mixed Fourier cosine series of
$f(x)=x-L$ on $[0,L]$.
\end{example}

\begin{solution}
 The coefficients are
\begin{align*}
c_n&=\frac{2}{L}\int_0^L(x-L)\cos\frac{(2n-1)\pi x}{2L}\,dx\\
&=\frac{4}{(2n-1)\pi}\left[(x-L)\sin\frac{(2n-1)\pi x}{2L}\lims0L-\int_0^L
\sin\frac{(2n-1)\pi x}{2L}\,dx\right]\\
&=\frac{8L}{(2n-1)^2\pi^2}
\cos\frac{(2n-1)\pi x}{2L}\lims0L
=-\frac{8L}{(2n-1)^2\pi^2}.
\end{align*}
Therefore
\[
C_M(x)=-\frac{8L}{\pi^2}\sum_{n=1}^\infty\frac{1}{(2n-1)^2}
\cos\frac{(2n-1)\pi x}{2L}.
\]
\zcref[S]{thmtype:11.3.3} implies that
\[
C_M(x)= x-L,\quad 0\le x\le L.\qedhere
\]
\end{solution}

\[
 \sin\frac{\pi x}{2L},
\, \sin\frac{3\pi x}{2L},\dots,
 \, \sin\frac{(2n-1)\pi x}{2L},\dots
\]
of the boundary value problem
\[
y''+\lambda y=0,\quad y(0)=0,\quad y'(L)=0
\]
(Problem~3)   are orthogonal on $[0,L]$.
 If $f$ is integrable on $[0,L]$, then the
Fourier expansion of $f$ in terms of these functions
is
\[
\sum_{n=1}^\infty d_n\sin\frac{(2n-1)\pi x}{2L},
\]
where
\[
d_n=\frac{\dst\int_0^Lf(x)\sin\frac{(2n-1)\pi x}{2L}\,dx}{\dst\int_0^L
\sin^2\frac{(2n-1)\pi x}{2L}\,dx}=\frac{2}{L}\int_0^Lf(x)\sin\frac{(2n-1)\pi
x}{2L}\,dx.

We'll  call this expansion the \emph{\color{blue}mixed Fourier sine
series} of $f$ on $[0,L]$.


It can be shown (\zcref{exer:11.3.58}) that
 the mixed Fourier sine series
of $f$ on $[0,L]$
is simply the restriction to  $[0,L]$ of the Fourier
sine series of

f_4(x)=
\begin{cases}
f(x),&0\le x\le L,\\f(2L-x),&L< x\le 2L,
\end{cases}
\]
on $[0,2L]$ (\zcref{figure:11.3.4}).

\begin{theorem}\color{blue} \label{thmtype:11.3.4}
If $f$ is piecewise smooth on $[0,L]$, then the mixed Fourier
sine series
\[
S_M(x)=\sum_{n=1}^\infty d_n\sin\frac{(2n-1)\pi x}{2L}
\]
of $f$ on $[0,L]$, with
\[
d_n=\frac{2}{L}\int_0^Lf(x)\sin\frac{(2n-1)\pi x}{2L}\,dx,
\]
 converges for all $x$ in $[0,L];    $ moreover,
\[
S_M(x)=
\begin{cases}
0&\text{ if $x=0$}  \\
f(x)&\text{ if $0<x<L$ and $f$ is continuous at $x$  }\\
\dfrac{f(x-)+f(x+)}{2}&\text{ if $0<x<L$ and $f$ is discontinuous at $x$}\\
f(L-)&\text{ if $x=L$.}
\end{cases}
\]
\end{theorem}


\begin{example}\label{example:11.3.4}
Find the mixed Fourier sine series of $f(x)=x$ on $[0,L]$.
\end{example}

\begin{solution}
 The coefficients are
\begin{align*}
d_n&=\frac{2}{L}\int_0^Lx\sin\frac{(2n-1)\pi x}{2L}\,dx\\
&=-\frac{4}{(2n-1)\pi}\left[x\cos\frac{(2n-1)\pi x}{2L}\lims0L-
\int_0^L
\cos\frac{(2n-1)\pi x}{2L}\,dx\right]\\
&=\frac{4}{(2n-1)\pi}
\int_0^L
\cos\frac{(2n-1)\pi x}{2L}\,dx\\
&=\frac{8L}{(2n-1)^2\pi^2}\sin\frac{(2n-1)\pi x}{2L}\lims0L=(-1)^{n+1}\frac{8L}{(2n-1)^2\pi^2}.
\end{align*}
Therefore
\[
S_M(x)=-\frac{8L}{\pi^2}\sum_{n=1}^\infty\frac{(-1)^n}{(2n-1)^2}
\sin\frac{(2n-1)\pi x}{2L}.
\]
\zcref[S]{thmtype:11.3.4} implies that
\[
S_M(x)=x,\quad 0\le x\le L.\qedhere
\]
\end{solution}


\boxit{A Useful Observation}

In applications involving expansions in
terms of the eigenfunctions of Problems~1--4, the functions being
expanded
 are often  polynomials that  satisfy the boundary conditions
of the problem under consideration. In this case the next theorem
presents an
 efficient way  to obtain the
coefficients in the expansion.

\begin{theorem}\label{thmtype:11.3.5}\begin{alist}[above=-\baselineskip]
\item
If $f'(0)=f'(L)=0$, $f''$ is continuous, and $f'''$ is piecewise
continuous  on $[0,L]$, then
\begin{equation} \label{eq:11.3.3}
f(x)=a_0+\sum_{n=1}^\infty a_n\cos\frac{n\pi x}{L},
\quad 0\le x\le L,
\end{equation}
with
\begin{equation} \label{eq:11.3.4}
a_0=\frac{1}{L}\int_0^L f(x)\,dx \quad\text{and}\quad
a_n= \frac{2L^2}{n^3\pi^3}\int_0^L f'''(x)\sin\frac{n\pi
x}{L}\,dx,
\quad n\ge1.
\end{equation}

Now suppose $f'$ is continuous and $f''$ is piecewise
continuous on $[0,L]$.

\item
If  $f(0)=f(L)=0$, then
\[
f(x)=\sum_{n=1}^\infty b_n\sin\frac{n\pi x}{L},
\quad 0\le x\le L,
\]
with
\begin{equation} \label{eq:11.3.5}
b_n=-\frac{2L}{n^2\pi^2}\int_0^L f''(x)\sin\frac{n\pi
x}{L}\,dx.
\end{equation}

\item
If  $f'(0)=f(L)=0$, then
\[
f(x)=
\sum_{n=1}^\infty c_n\cos\frac{(2n-1)\pi x}{2L},
\quad 0\le x\le L,
\]
with
\begin{equation} \label{eq:11.3.6}
c_n=-\frac{8L}{(2n-1)^2\pi^2}\int_0^L f''(x)\cos\frac{(2n-1)\pi x}{2L}
\,dx.
\end{equation}

\item
If  $f(0)=f'(L)=0$, then
\[
f(x)=
\sum_{n=1}^\infty d_n\sin\frac{(2n-1)\pi x}{2L},
\quad 0\le x\le L,
\]
with
\begin{equation} \label{eq:11.3.7}
d_n=-\frac{8L}{(2n-1)^2\pi^2}\int_0^L f''(x)\sin\frac{(2n-1)\pi x}{2L}
\,dx.
\end{equation}
\end{alist}
\end{theorem}


\begin{proof}
We'll prove (a) and leave the rest to you
(\zcref{exer:11.3.35,exer:11.3.42,exer:11.3.50}).  Since
$f$ is continuous on
$[0,L]$, \zcref{thmtype:11.3.1} implies \eqref{eq:11.3.3} with $a_0$,
$a_1$, $a_2$,... as defined in \zcref{thmtype:11.3.1}. We already
know that
$a_0$ is as in
\eqref{eq:11.3.4}. If  $n\ge1$, integrating twice by parts yields
\begin{align*}
a_n&=\frac{2}{L}\int_0^L f(x)\cos\frac{n\pi x}{L}\,dx\\
&=\frac{2}{n\pi}\left[f(x)\sin\frac{n\pi x}{L}\lims0L
-\int_0^Lf'(x)\sin\frac{n\pi x}{L}\,dx\right]\\
&=-\frac{2}{n\pi}
\int_0^Lf'(x)\sin\frac{n\pi x}{L}\,dx
\qquad\text{(since $\sin0=\sin n\pi=0$)}\\
&=\frac{2L}{n^2\pi^2}\left[f'(x)\cos\frac{n\pi x}{L}\lims0L
-\int_0^Lf''(x)\cos\frac{n\pi x}{L}\right]\,dx\\
&=-\frac{2L}{n^2\pi^2}\int_0^Lf''(x)\cos\frac{n\pi x}{L}\,dx
\qquad\text{(since $f'(0)=f'(L)=0$)}\\
&=-\frac{2L^2}{n^3\pi^3}\left[f''(x)\sin\frac{n\pi x}{L}\lims0L
-\int_0^Lf'''(x)\sin\frac{n\pi x}{L}\,dx\right]\\
&=
\frac{2L^2}{n^3\pi^3}\int_0^Lf'''(x)\sin\frac{n\pi x}{L}\,dx
\qquad\text{(since $\sin0=\sin n\pi=0$).}
\end{align*}
(By an argument similar to one used in the proof of \zcref{thmtype:8.3.1},
 the last integration by parts is legitimate
in the case where $f'''$ is undefined at finitely many points in
$[0,L]$, so long as it's piecewise continuous on $[0,L]$.)
This completes the proof.
\end{proof}


\begin{example}\label{example:11.3.5}
Find the Fourier cosine expansion of $f(x)=x^2(3L-2x)$
on $[0,L]$.
\end{example}

\begin{solution}
Here
\[
a_0=\frac{1}{L}\int_0^L(3Lx^2-2x^3)\,dx=\frac{1}{L}\left(Lx^3-\frac{x^4}{2}
\right)\lims0L=\frac{L^3}{2}
\]
and
\[
a_n=\frac{2}{L}\int_0^L(3Lx^2-2x^3)\cos\frac{n\pi x}{L}\,dx,\quad n\ge1.
\]
Evaluating this integral directly is laborious.
However, since $f'(x)=6Lx-6x^2$, we see that $f'(0)=f'(L)=0$.
Since $f'''(x)=-12$, we see from \eqref{eq:11.3.4} that if $n\ge1$ then
\begin{align*}
a_n&=-\frac{24L^2}{n^3\pi^3}\int_0^L\sin\frac{n\pi x}{L}\,dx
=\frac{24L^3}{n^4\pi^4}\cos\frac{n\pi x}{L}\lims0L=\frac{24L^3}{n^4\pi^4}[(-1)^n-1]\\
&=
\begin{cases}
-\dfrac{48L^3}{(2m-1)^4\pi^4}&\text{ if  $n=2m-1$},\\
0&\text{ if $n=2m$.}
\end{cases}
\end{align*}
Therefore
\[
C(x)=\frac{L^3}{2}-\frac{48L^3}{\pi^4}\sum_{n=1}^\infty\frac{1}{(2n-1)^4}\cos\frac{(2n-1)\pi x}{L}.\qedhere
\]
\end{solution}

\begin{example}\label{example:11.3.6}
Find the Fourier sine expansion of
$f(x)=x(x^2-3Lx+2L^2)$
on $[0,L]$.
\end{example}

\begin{solution}
Since $f(0)=f(L)=0$ and $f''(x)=6(x-L)$, we see from \eqref{eq:11.3.5}
that
\begin{align*}
b_n&=-\frac{12L}{n^2\pi^2}\int_0^L(x-L)\sin\frac{n\pi x}{L}\,dx\\
&=\frac{12L^2}{n^3\pi^3}\left[(x-L)\cos\frac{n\pi x}{L}\lims0L
-\int_0^L\cos\frac{n\pi x}{L}\,dx\right]\\
&=\frac{12L^2}{n^3\pi^3}\left[L-\frac{L}{n\pi}\sin\frac{n\pi
x}{L}\lims0L\right]
=\frac{12L^3}{n^3\pi^3}.
\end{align*}
Therefore
\[
S(x)=\frac{12L^3}{\pi^3}\sum_{n=1}^\infty\frac{1}{n^3}
\sin\frac{n\pi x}{L}.\qedhere
\]
\end{solution}

\begin{example}\label{example:11.3.7}
Find  the mixed Fourier cosine expansion of
$f(x)=3x^3-4Lx^2+L^3$
on $[0,L]$.
\end{example}

\begin{solution}
Since $f'(0)=f(L)=0$ and $f''(x)=2(9x-4L)$, we see from \eqref{eq:11.3.6}
that
\begin{align*}
c_n&=
-\frac{16L}{(2n-1)^2\pi^2}
\int_0^L(9x-4L)\cos\frac{(2n-1)\pi x}{2L}\,dx\\
&=-\frac{32L^2}{(2n-1)^3\pi^3}\left[(9x-4L)\sin\frac{(2n-1)\pi x}{2L}
\lims0L-9\int_0^L\sin\frac{(2n-1)\pi x}{2L}\right]\,dx\\
&=-\frac{32L^2}{(2n-1)^3\pi^3}
\left[(-1)^{n+1}5L+\frac{18L}{(2n-1)\pi}\cos\frac{(2n-1)\pi x}{2L}
\lims0L\right] \\
&=\frac{32L^3}{(2n-1)^3\pi^3}
\left[(-1)^n5+\frac{18}{(2n-1)\pi}\right].
\end{align*}
Therefore
\[
C_M(x)=\frac{32L^3}{\pi^3}\sum_{n=1}^\infty\frac{1}{(2n-1)^3}\left[
(-1)^n5+\frac{18}{(2n-1)\pi}\right]\cos\frac{(2n-1)\pi
x}{2L}.\qedhere
\]
\end{solution}

\begin{example}\label{example:11.3.8}
Find  the mixed Fourier  sine expansion of
\[
f(x)=x(2x^2-9Lx+12L^2)
\]
on $[0,L]$.
\end{example}

\begin{solution}
Since $f(0)=f'(L)=0$, and $f''(x)=6(2x-3L)$, we see from \eqref{eq:11.3.7}
that
\begin{align*}
d_n&=-\frac{48L}{(2n-1)^2\pi^2}\int_0^L(2x-3L)
\sin\frac{(2n-1)\pi x}{2L}\,dx\\
&=\frac{96L^2}{(2n-1)^3\pi^3}\left[(2x-3L)\cos\frac{(2n-1)\pi
x}{2L}\lims0L-2\int_0^L\cos\frac{(2n-1)\pi x}{2L}\,dx\right]\\
&=\frac{96L^2}{(2n-1)^3\pi^3}\left[3L-\frac{4L}{(2n-1)\pi}
\sin\frac{(2n-1)\pi x}{2L}\lims0L\right]\\
&=\frac{96L^3}{(2n-1)^3\pi^3}\left[3+(-1)^n\frac{4}{(2n-1)\pi}\right].
\end{align*}
Therefore
\[
S_M(x)
=\frac{96L^3}{\pi^3}\sum_{n=1}^\infty
\frac{1}{(2n-1)^3}\left[3+(-1)^n\frac{4}{(2n-1)\pi}\right]\sin\frac{(2n-1)\pi
x}{2L}.\qedhere
\]
\end{solution}
